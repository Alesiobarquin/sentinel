"""Continue real captures with an explicit international-checkout workload.

Developer-only local HTTP traffic, outside the investigation tool layer.
"""
from pathlib import Path
import json
import shutil
import sys
import threading
import time
import urllib.request
import uuid

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))
from evals import benchmark as b
from evals.corpus import catalog, digest, write_json

COHORT = ROOT / 'var/resume-evaluation/20261005-fixed'
original_checks, original_write = b.preparation_checks, b.write_json


def shipping_checks(case, corpus):
    initial = original_checks(case, corpus)
    if case['id'] != 'shipping-slowdown':
        return initial
    records = corpus['records']
    native = next(r['raw'] for key, r in records.items() if json.loads(key) == {
        'name': 'traces', 'service': 'shipping', 'period': 'incident'})
    window = native['window']
    spans = [s for t in native['traces'] for s in t['spans']]
    observed = any(s['service'] == 'shipping' and s['operation'] == 'POST /ship-order' and
                   s['duration_ms'] >= 5000 and window['start'] <= s['start_time'] <= window['end'] for s in spans)
    checks = {k: v for k, v in initial['checks'].items() if k != 'real_backend_marker_present'}
    checks['real_delayed_shipping_span'] = observed
    return {'checks': checks, 'passed': all(checks.values()), 'original_text_check': initial,
            'interpretation': 'International checkout traffic exercises the real five-second shipping branch. Shipping logs are not exported to OpenSearch in this target; readiness requires a real in-window delayed shipping server span. This validates setup, not AI diagnosis.'}


def publish_ready(path, value):
    original_write(path, value)
    if path.name == 'ready.json':
        target = COHORT / path.parent.name
        if target.exists():
            raise ValueError('Never overwrite an existing accepted capture')
        shutil.copytree(path.parent, target)
        print('COHORT READY', path.parent.name, flush=True)


def international_checkout(stop):
    people = json.loads((ROOT / '.cache/demo/source/src/load-generator/people.json').read_text())
    person = next(p for p in people if p['address']['country'] == 'Canada')
    audit = COHORT / 'international-workload.jsonl'
    while not stop.is_set():
        user = str(uuid.uuid4())
        checkout = {**person, 'userId': user}
        cart = {'userId': user, 'item': {'productId': 'OLJCESPC7Z', 'quantity': 1}}
        for route, body in [('/api/cart', cart), ('/api/checkout', checkout)]:
            started = time.monotonic()
            status = None
            error = None
            try:
                req = urllib.request.Request('http://127.0.0.1:8080' + route, data=json.dumps(body).encode(),
                    headers={'Content-Type': 'application/json', 'baggage': 'synthetic_request=true'}, method='POST')
                with urllib.request.urlopen(req, timeout=25) as response:
                    status = response.status
                    response.read(1_000_001)
            except Exception as exc:
                error = type(exc).__name__
            with audit.open('a') as stream:
                stream.write(json.dumps({'at': time.time(), 'route': route, 'country': 'Canada',
                    'status': status, 'error_type': error, 'latency_ms': (time.monotonic()-started)*1000})+'\n')
        stop.wait(5)


b.preparation_checks = shipping_checks
b.write_json = publish_ready
failed = COHORT / 'shipping-slowdown'
archive = COHORT / 'failed-preparations/shipping-default-traffic'
if failed.exists():
    archive.parent.mkdir(exist_ok=True)
    if archive.exists():
        raise ValueError('Failed preparation archive already exists')
    failed.rename(archive)
write_json(COHORT / 'workload-method.json', {'script_sha256': digest(Path(__file__).read_bytes()),
    'shipping_workload': 'Real frontend add-to-cart and checkout requests with the upstream Canadian fixture; identical driver during baseline and incident; synthetic_request baggage; local URL only.',
    'initial_failed_capture': 'Shipping flag was enabled but sampled domestic traffic did not exercise the international branch; no model used that capture.',
    'shipping_log_limit': 'Docker stdout exists but shipping log data is not exported to OpenSearch; readiness uses actual delayed server spans.'})
stop = threading.Event()
thread = threading.Thread(target=international_checkout, args=(stop,), daemon=True)
thread.start()
try:
    b.prepare(COHORT / '_shipping-recapture', seconds=180, selected=['shipping-slowdown'])
finally:
    stop.set()
    thread.join(timeout=30)
if thread.is_alive():
    raise ValueError('Workload thread did not stop; do not continue another case')
remaining = [case['id'] for case in catalog() if not (COHORT / case['id'] / 'ready.json').exists()]
if remaining:
    b.prepare(COHORT / '_remaining-captures', seconds=180, selected=remaining)
write_json(COHORT / 'capture-complete.json', {'scenarios': len(catalog()), 'completed_at': time.time(),
                                          'initial_failed_preparations_retained': 1})
